
#include <Arduino.h>
#include <Wire.h>
#include <SPI.h>
#include <SD.h>

#include <RTClib.h>
#include <Adafruit_INA219.h>

// ================= PIN CONFIG =================

// I2C
#define I2C_SDA 21
#define I2C_SCL 22

// SD SPI
#define SD_SCK   18
#define SD_MISO  19
#define SD_MOSI  23
#define SD_CS     5

// ================= OBJECTS =================

RTC_DS3231 rtc;
Adafruit_INA219 ina219;

SPIClass sdSPI(VSPI);

// ================= STATUS =================

bool rtcReady = false;
bool inaReady = false;
bool sdReady = false;

const char *LOG_FILE = "/test_log.csv";

unsigned long previousMillis = 0;
const unsigned long LOG_INTERVAL = 2000;

// ================= TIME FORMAT =================

String getTimestamp() {
    DateTime now = rtc.now();

    char buffer[25];

    snprintf(buffer, sizeof(buffer),
             "%04d-%02d-%02d %02d:%02d:%02d",
             now.year(),
             now.month(),
             now.day(),
             now.hour(),
             now.minute(),
             now.second());

    return String(buffer);
}

// ================= SD INIT =================

void initSD() {

    Serial.println("\n[SD] Initializing...");

    sdSPI.begin(SD_SCK, SD_MISO, SD_MOSI, SD_CS);

    if (!SD.begin(SD_CS, sdSPI, 10000000)) {
        Serial.println("[SD] Initialization FAILED!");
        return;
    }

    sdReady = true;

    Serial.println("[SD] Initialization SUCCESS!");

    if (!SD.exists(LOG_FILE)) {

        File file = SD.open(LOG_FILE, FILE_WRITE);

        if (file) {
            file.println("Timestamp,Bus_Voltage_V,Shunt_mV,Current_mA,Power_mW");
            file.close();

            Serial.println("[SD] CSV file created.");
        } else {
            Serial.println("[SD] File creation FAILED!");
        }
    }
}

// ================= RTC INIT =================

void initRTC() {

    Serial.println("\n[RTC] Initializing...");

    if (!rtc.begin()) {
        Serial.println("[RTC] DS3231 NOT FOUND!");
        return;
    }

    rtcReady = true;

    Serial.println("[RTC] DS3231 detected.");

    if (rtc.lostPower()) {

        Serial.println("[RTC] Time lost! Setting compile time.");

        rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    }

    Serial.print("[RTC] Current time: ");
    Serial.println(getTimestamp());
}

// ================= INA219 INIT =================

void initINA219() {

    Serial.println("\n[INA219] Initializing...");

    if (!ina219.begin()) {
        Serial.println("[INA219] NOT FOUND!");
        return;
    }

    inaReady = true;

    // Default INA219 calibration
    ina219.setCalibration_32V_2A();

    Serial.println("[INA219] Detected!");
}

// ================= SENSOR READ =================

void readINA219() {

    if (!inaReady) return;

    float busVoltage = ina219.getBusVoltage_V();
    float shuntVoltage = ina219.getShuntVoltage_mV();
    float current = ina219.getCurrent_mA();
    float power = ina219.getPower_mW();

    Serial.println("\n========== INA219 ==========");

    Serial.printf("Bus Voltage  : %.3f V\n", busVoltage);
    Serial.printf("Shunt Voltage: %.3f mV\n", shuntVoltage);
    Serial.printf("Current      : %.3f mA\n", current);
    Serial.printf("Power        : %.3f mW\n", power);

    Serial.println("============================");

    if (sdReady && rtcReady) {

        File file = SD.open(LOG_FILE, FILE_APPEND);

        if (file) {

            file.printf("%s,%.3f,%.3f,%.3f,%.3f\n",
                        getTimestamp().c_str(),
                        busVoltage,
                        shuntVoltage,
                        current,
                        power);

            file.close();

            Serial.println("[SD] Data saved.");
        } else {
            Serial.println("[SD] Write FAILED!");
        }
    }
}

// ================= SETUP =================

void setup() {

    Serial.begin(115200);
    delay(1000);

    Serial.println("\n\n================================");
    Serial.println(" ESP32 MODULE TEST");
    Serial.println(" SD CARD + DS3231 + INA219");
    Serial.println("================================");

    Wire.begin(I2C_SDA, I2C_SCL);

    initRTC();
    initINA219();
    initSD();

    Serial.println("\n========== MODULE STATUS ==========");

    Serial.printf("RTC    : %s\n", rtcReady ? "OK" : "FAILED");
    Serial.printf("INA219 : %s\n", inaReady ? "OK" : "FAILED");
    Serial.printf("SD     : %s\n", sdReady ? "OK" : "FAILED");

    Serial.println("===================================");
}

// ================= LOOP =================

void loop() {

    if (millis() - previousMillis >= LOG_INTERVAL) {

        previousMillis = millis();

        if (rtcReady) {
            Serial.print("\nTimestamp: ");
            Serial.println(getTimestamp());
        }

        readINA219();
    }
}
